```gleam
// bleam/src/bleam/config.gleam
import gleam/json
import gleam/dynamic

pub type Config {
  Config(model_type: String, vocab_size: Int)
}

pub fn from_json(json_string: String) -> Result(Config, json.DecodeError) {
  let decoder = dynamic.decode2(
    Config,
    dynamic.field("model_type", dynamic.string),
    dynamic.field("vocab_size", dynamic.int),
  )
  
  json.decode(from: json_string, using: decoder)
}
```
